drop database if exists IInstitute;
create database if not exists IInstitute;

USE IInstitute;

-- Create Students table
CREATE TABLE Students (
    StudentID INT AUTO_INCREMENT PRIMARY KEY,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    Gender VARCHAR(10),  -- changed from ENUM to VARCHAR
    DOB DATE,
    Email VARCHAR(100) UNIQUE,
    Phone VARCHAR(15)
);

-- Create Courses table
CREATE TABLE Courses (
    CourseID INT AUTO_INCREMENT PRIMARY KEY,
    CourseName VARCHAR(100),
    DurationMonths INT,
    Fee DECIMAL(10,2)
);

-- Create Enrollments table
CREATE TABLE Enrollments (
    EnrollmentID INT AUTO_INCREMENT PRIMARY KEY,
    StudentID INT,
    CourseID INT,
    EnrollmentDate DATE,
    FOREIGN KEY (StudentID) REFERENCES Students(StudentID),
    FOREIGN KEY (CourseID) REFERENCES Courses(CourseID)
);

create view StudentCourseDetails as
select 
	concat(s.FirstName,' ',  s.LastName) as Fullname,
	c.courseName,
	c.Fee,
	e.EnrollmentDate
from students s
join enrollments e on s.StudentID = e.StudentID
join courses c on e.CourseID = c.CourseID;

DELIMITER //
CREATE FUNCTION GetStudentCourseCount(p_StudentID INT)
RETURNS INT
DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE course_count INT;

    SELECT COUNT(*)
    INTO course_count
    FROM enrollments
    WHERE StudentID = p_StudentID;

    RETURN course_count;
END //
DELIMITER ;

DELIMITER //

CREATE PROCEDURE GetStudentCourses(IN p_StudentID INT)
BEGIN
    SELECT
        CONCAT(s.FirstName, ' ', s.LastName) AS StudentFullName,
        c.CourseName,
        c.DurationMonths AS CourseDuration,
        c.Fee AS CourseFee,
        e.EnrollmentDate
    FROM students s
    JOIN enrollments e ON s.StudentID = e.StudentID
    JOIN courses c ON e.CourseID = c.CourseID
    WHERE s.StudentID = p_StudentID;
END //

DELIMITER ;

CALL GetStudentCourses(3);




DELIMITER //

CREATE PROCEDURE GetCourseEnrollmentCount(
    IN p_CourseID INT,
    OUT p_TotalStudents INT
)
BEGIN
    SELECT COUNT(*)
    INTO p_TotalStudents
    FROM enrollments
    WHERE CourseID = p_CourseID;
END //

DELIMITER ;
CALL GetCourseEnrollmentCount(4, @TotalStudents);
